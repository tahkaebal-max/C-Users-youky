class CoordinateSystem
{
    static void Main()
    {
        Position2D pos1 = new Position2D();
        pos1.Horizontal = 9;
        pos1.Vertical = 8;
        pos1.ShowCoordinates();

        Position2D pos2 = new Position2D();
        pos2.Horizontal = 12;
        pos2.Vertical = 16;
        pos2.ShowCoordinates();

        double separation = pos1.MeasureSeparation(pos2);
        Console.WriteLine($"Дистанция: {separation}");

        pos1.Translate(4, -2);
        pos1.ShowCoordinates();
    }
}

class Position2D
{
    public double Horizontal;
    public double Vertical;

    public void ShowCoordinates()
    {
        Console.WriteLine($"Позиция: [X: {Horizontal}, Y: {Vertical}]");
    }

    public double MeasureSeparation(Position2D target)
    {
        double diffX = Horizontal - target.Horizontal;
        double diffY = Vertical - target.Vertical;
        return Math.Sqrt(diffX * diffX + diffY * diffY);
    }

    public void Translate(double shiftX, double shiftY)
    {
        Horizontal += shiftX;
        Vertical += shiftY;
        Console.WriteLine($"Смещение: (по X: {shiftX}, по Y: {shiftY})");
    }
}
